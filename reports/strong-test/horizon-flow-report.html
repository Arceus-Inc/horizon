import pandas as pd, sqlite3

con=sqlite3.connect('warehouse.db')

sql="""WITH agg_sales AS (

  SELECT region, quarter, SUM(revenue) AS revenue

  FROM sales

  GROUP BY region, quarter

),

agg_costs AS (

  SELECT region, quarter, SUM(cost) AS cost

  FROM costs

  GROUP BY region, quarter

),

keys AS (

  SELECT region, quarter FROM agg_sales

  UNION

  SELECT region, quarter FROM agg_costs

)

SELECT

  k.region AS region,

  k.quarter AS quarter,

  COALESCE(s.revenue, 0) AS revenue,

  COALESCE(c.cost, 0) AS cost,

  COALESCE(s.revenue, 0) - COALESCE(c.cost, 0) AS profit,

  CASE

    WHEN COALESCE(s.revenue, 0) = 0 THEN NULL

    ELSE (COALESCE(s.revenue, 0) - COALESCE(c.cost, 0)) * 1.0 / COALESCE(s.revenue, 0)

  END AS profit_margin

FROM keys k

LEFT JOIN agg_sales s

  ON s.region = k.region AND s.quarter = k.quarter

LEFT JOIN agg_costs c

  ON c.region = k.region AND c.quarter = k.quarter

ORDER BY k.region, k.quarter;"""

df=pd.read_sql_query(sql, con)

#   We treat those as the Q1/Q2/Q3 buckets.
#
# Momentum definition (Q1→Q3):
#   Fit a straight line to quarterly profit totals for Q1,Q2,Q3 where x=[1,2,3]
#   and y=[profit_Q1, profit_Q2, profit_Q3].
#   Momentum = slope of the least-squares line (profit per quarter).
#   With equally spaced quarters and 3 points, slope simplifies to (profit_Q3 - profit_Q1)/2.


def compute(db_path: str = DB_PATH):
    con = sqlite3.connect(db_path)
    cur = con.cursor()

    cur.execute(
        """
        SELECT s.region, s.quarter, s.revenue, c.cost
        FROM sales s
        JOIN costs c
          ON s.region = c.region AND s.quarter = c.quarter
        """
    )
    rows = cur.fetchall()
    con.close()

    # Aggregate profit by region/quarter (defensive even though data is 1 row per region-quarter)
    profit = defaultdict(lambda: defaultdict(float))
    regions = set()
    for region, quarter, revenue, cost in rows:
        regions.add(region)
        profit[region][quarter] += float(revenue) - float(cost)

    out = []
    for region in sorted(regions):
        q1 = float(profit[region].get("Q1", 0.0))
        q2 = float(profit[region].get("Q2", 0.0))
        q3 = float(profit[region].get("Q3", 0.0))

        delta = q3 - q1
        growth = delta / abs(q1) if q1 != 0 else float("inf")
        momentum = (q3 - q1) / 2.0

        out.append(
            {
                "region": region,
                "Q1": q1,
                "Q2": q2,
                "Q3": q3,
                "delta_Q1_Q3": delta,
                "growth_Q1_Q3": growth,
                "momentum": momentum,
            }
        )

    # Rank: highest momentum first; tie-break by region ascending for determinism
    out_sorted = sorted(out, key=lambda r: (-r["momentum"], r["region"]))
    for i, r in enumerate(out_sorted, start=1):
        r["momentum_rank"] = i

    return out_sorted
