import sqlite3

DB_PATH = "warehouse.db"
OUT_METRICS = "region_quarter_metrics.csv"
OUT_DECOMP = "profit_decomposition_q1_q3.csv"

QUARTERS = ["Q1", "Q2", "Q3"]


def fetch_joined_rows(db_path: str):
    con = sqlite3.connect(db_path)
    con.row_factory = sqlite3.Row
    cur = con.cursor()
    cur.execute(
        """
        SELECT s.region, s.quarter, s.revenue, s.units, c.cost
        FROM sales s
        JOIN costs c USING(region, quarter)
        WHERE s.quarter IN ('Q1','Q2','Q3')
        ORDER BY s.region, s.quarter;
        """
    )
    rows = [dict(r) for r in cur.fetchall()]
    con.close()
    return rows


def compute_metrics(rows):
    # Add derived metrics per (region, quarter)
    for r in rows:
        r["profit"] = r["revenue"] - r["cost"]
        r["asp"] = r["revenue"] / r["units"]
        r["unit_cost"] = r["cost"] / r["units"]
    return rows


def index_by_region_quarter(rows):
    return {(r["region"], r["quarter"]): r for r in rows}


def overall_by_quarter(rows):
    overall = {q: {"region": "Overall", "quarter": q, "revenue": 0, "units": 0, "cost": 0} for q in QUARTERS}
    for r in rows:
        o = overall[r["quarter"]]
        o["revenue"] += r["revenue"]
        o["units"] += r["units"]
        o["cost"] += r["cost"]
    for q in QUARTERS:
        o = overall[q]
        o["profit"] = o["revenue"] - o["cost"]
        o["asp"] = o["revenue"] / o["units"]
        o["unit_cost"] = o["cost"] / o["units"]
    return overall


def profit_decomposition(q1, q3):
    """Additive decomposition of profit change using identity:

    profit = units * (asp - unit_cost)

    Let Q1 values be U1,P1,C1; Q3 be U3,P3,C3.

    Δprofit = (U3-U1)*(P1-C1) + U3*(P3-P1) - U3*(C3-C1)

    Returns components that sum to Δprofit (up to floating rounding).
    """

    U1, U3 = q1["units"], q3["units"]
    P1, P3 = q1["asp"], q3["asp"]
    C1, C3 = q1["unit_cost"], q3["unit_cost"]

    units_driver = (U3 - U1) * (P1 - C1)
    price_driver = U3 * (P3 - P1)
    cost_driver = -U3 * (C3 - C1)

    return {
        "units_driver": units_driver,
        "price_driver": price_driver,
        "cost_driver": cost_driver,
        "sum_drivers": units_driver + price_driver + cost_driver,
    }


def write_csv(path, header, rows):
    with open(path, "w", newline="", encoding="utf-8") as f:
        f.write(",".join(header) + "\n")
        for r in rows:
            f.write(",".join(str(r.get(h, "")) for h in header) + "\n")

def fetch_region_baseline_q3(db_path: str = DB_PATH):
    """Return dict region -> baseline metrics using Q3 as 'next-quarter' proxy.

    Baseline profit in warehouse is revenue - cost; we use contribution margin fraction:
      margin = (revenue - cost) / revenue
    """
    con = sqlite3.connect(db_path)
    cur = con.cursor()
    cur.execute(
        """
        SELECT s.region, s.revenue, c.cost
        FROM sales s
        JOIN costs c USING(region, quarter)
        WHERE s.quarter = 'Q3'
        ORDER BY s.region;
        """
    )
    rows = cur.fetchall()
    con.close()

    regions = {}
    for region, revenue, cost in rows:
        revenue = float(revenue)
        cost = float(cost)
        if revenue <= 0:
            raise ValueError(f"Non-positive revenue for region {region} in Q3")
        margin = (revenue - cost) / revenue
        regions[region] = {
            "region": region,
            "baseline_revenue": revenue,
            "baseline_cost": cost,
            "baseline_profit": revenue - cost,
            "margin": margin,
        }
    return regions


def incremental_profit(baseline_revenue: float, margin: float, uplift_pct: float, var_cost_pct: float, fixed_invest: float) -> float:
    """Incremental profit model.

    Let:
      R0 = baseline revenue
      m  = gross margin fraction on revenue (= (R0 - C0)/R0)
      u  = revenue uplift fraction from investment
      vc = variable cost fraction on baseline revenue (incurred if we invest)
      F  = fixed investment

    Incremental profit:
      IP = (R0 * u) * m  - (R0 * vc) - F

    Interprets uplift as incremental revenue with same margin m.
    Variable cost is modeled as a % of baseline revenue to keep it observable/region-scaled.
    """
    return (baseline_revenue * uplift_pct) * margin - (baseline_revenue * var_cost_pct) - fixed_invest

def write_region_breakeven(regions, scenarios, out_path=OUT_BREAKEVEN):
    """Break-even analyses per region.

    1) Break-even uplift_pct given base scenario (vc, F):
         0 = R0*u*m - R0*vc - F  =>  u* = (R0*vc + F)/(R0*m)

    2) Max fixed investment F* given base scenario (u, vc):
         F* = R0*u*m - R0*vc

    3) Max variable cost pct vc* given base scenario (u, F):
         vc* = u*m - F/R0
    """
    base = next(s for s in scenarios if s["scenario"] == "base")

    rows = []
    for r in regions.values():
        R0 = r["baseline_revenue"]
        m = r["margin"]
        u = base["uplift_pct"]
        vc = base["var_cost_pct"]
        F = base["fixed_invest"]

        if m <= 0:
            u_star = math.inf
        else:
            u_star = (R0 * vc + F) / (R0 * m)

        F_star = R0 * u * m - R0 * vc
        vc_star = u * m - (F / R0)

        rows.append(
            {
                "region": r["region"],
                "baseline_revenue": R0,
                "margin": m,
                "base_uplift_pct": u,
                "base_var_cost_pct": vc,
                "base_fixed_invest": F,
                "breakeven_uplift_pct": u_star,
                "max_fixed_invest": F_star,
                "max_var_cost_pct": vc_star,
            }
        )

    rows.sort(key=lambda x: x["region"])
    fieldnames = list(rows[0].keys())
    with open(out_path, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=fieldnames)
        w.writeheader()
        w.writerows(rows)


def compute_ranking_from_outputs(out_profit=OUT_PROFIT):
    # region -> expected value
    ev = {}
    with open(out_profit, newline="", encoding="utf-8") as f:
        r = csv.DictReader(f)
        for row in r:
            region = row["region"]
            ev[region] = float(row["expected_value_incremental_profit"])
    ranking = sorted(ev.items(), key=lambda kv: (-kv[1], kv[0]))
